## 路由机制

前端实现了一个简易的路由系统：

```javascript
// URL参数处理
function handleRouting() {
  const params = new URLSearchParams(window.location.search);
  
  // 处理标签页切换
  const tab = params.get('tab');
  if (tab === 'profile') {
    directSwitchToTab('profile');
  } else if (tab === 'create') {
    directSwitchToTab('create');
  }
  
  // 处理游戏详情
  const gameId = params.get('game');
  if (gameId) {
    openGame(gameId);
  }
  
  // 处理推荐码
  const ref = params.get('ref');
  if (ref) {
    handleReferralParams();
  }
}
```

### 支持的URL格式

```
/                     → 首页
/?tab=profile         → 我的页面
/?tab=create          → 创作页面
/?game=xxx            → 打开游戏详情
/?ref=xxx             → 推荐链接
/games.html?source=my-games     → 我的游戏列表
/games.html?source=my-favorites → 收藏列表
/games.html?source=my-likes     → 点赞列表
/games.html?category=益智       → 分类游戏列表
/games.html?sort=popular        → 按热度排序
```

```javascript
// 使用 IntersectionObserver
const imageObserver = new IntersectionObserver((entries) => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      const img = entry.target;
      img.src = img.dataset.src;
      imageObserver.unobserve(img);
    }
  });
});
```

### 6. 游戏加载优化

```javascript
// 使用 Blob URL 加载游戏，避免重复传输
function displayGame(code, title) {
  const blob = new Blob([code], { type: 'text/html' });
  const blobUrl = URL.createObjectURL(blob);
  gameFrame.src = blobUrl;
  
  // 加载完成后释放
  gameFrame.onload = () => {
    URL.revokeObjectURL(blobUrl);
  };
}
```
